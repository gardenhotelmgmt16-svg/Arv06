        function renderOTAReport() {
            const monthInput = document.getElementById('otaMonthFilter').value;
            if (!monthInput) return;
            const [year, month] = monthInput.split('-').map(Number);
            const otaBookingsForMonth = bookings.filter(b => {
                const date = new Date(b.checkIn);
                return OTA_LIST.includes(b.paymentMethod) && date.getFullYear() === year && date.getMonth() + 1 === month;
            });
            const summary = {}; OTA_LIST.forEach(ota => summary[ota] = { count: 0, revenue: 0 });
            let totalRevenue = 0;
            otaBookingsForMonth.forEach(b => { if (summary[b.paymentMethod]) { summary[b.paymentMethod].count++; summary[b.paymentMethod].revenue += b.total; totalRevenue += b.total; } });
            for (const key in summary) {
                const safeKey = key.replace(' ', '').replace('.', '').toLowerCase();
                const countEl = document.getElementById(`ota-${safeKey}-count`);
                const revEl = document.getElementById(`ota-${safeKey}-revenue`);
                if (countEl) countEl.textContent = `${summary[key].count} Booking`;
                if (revEl) revEl.textContent = formatCurrency(summary[key].revenue);
            }
            document.getElementById('ota-total-revenue').textContent = formatCurrency(totalRevenue);
            const selectedOta = document.getElementById('filterOtaDetail').value;
            let tableData = otaBookingsForMonth;
            if (selectedOta !== 'all') { tableData = otaBookingsForMonth.filter(b => b.paymentMethod === selectedOta); }
            const tbody = document.getElementById('otaTableBody');
            
            const formatDateID = (dateStr) => {
                const d = new Date(dateStr);
                return d.toLocaleDateString('id-ID', { day: '2-digit', month: '2-digit', year: 'numeric' });
            };

            tbody.innerHTML = tableData.length ? tableData.map(b => {
                const idx = bookings.findIndex(item => item.noBooking === b.noBooking && item.checkIn === b.checkIn); 
                return `<tr>
                    <td><span class="font-mono text-sm">${b.noBooking}</span></td>
                    <td>${b.guestName}</td>
                    <td>${formatDateID(b.checkIn)}</td>
                    <td>${formatDateID(b.checkOut)}</td>
                    <td><span class="badge bg-teal-100 text-teal-700">${b.paymentMethod}</span></td>
                    <td>${formatCurrency(b.price)}</td>

                </tr>`;
            }).join('') : '<tr><td colspan="8" class="text-center py-6 text-teal-500">Tidak ada transaksi</td></tr>';
        }

        function exportRecapToExcel() {

        // UPDATED: Room Rendering Logic
        function renderRooms(filter = 'all') { 
            const grid = document.getElementById('roomGrid'); grid.innerHTML = ''; 
            let selectedDate = document.getElementById('roomDateFilter').value;
            const today = formatDateInput(new Date());
            if (!selectedDate) { selectedDate = today; document.getElementById('roomDateFilter').value = today; }
            const filteredByType = filter === 'all' ? ROOMS : ROOMS.filter(r => r.type.toLowerCase() === filter); 
            [1, 2, 3].forEach(floor => { 
                const floorRooms = filteredByType.filter(r => r.floor === floor); 
                if (floorRooms.length === 0) return; 
                const label = document.createElement('div'); label.className = 'col-span-full mt-2 mb-1'; label.innerHTML = `<span class="text-xs font-bold text-teal-600">Lantai ${floor}</span>`; grid.appendChild(label); 
                floorRooms.forEach((room) => { 
                    let status = 'available';
                    let isManual = false;

                    // Check Manual Override first
                    if (manualRoomStatus[room.no]) { 
                        status = manualRoomStatus[room.no]; 
                        isManual = true; 
                    } else {
                        // UPDATED: Between Logic
                        // Occupied if: CheckIn <= Selected < CheckOut
                        // If CheckOut == Selected, room is FREE.
                        const occupied = bookings.find(b => b.roomNo === room.no && b.checkIn <= selectedDate && b.checkOut > selectedDate); 
                        if (occupied) { 
                            status = 'occupied'; 
                        } else if (selectedDate === today) { 
                            // Reserved logic: Only show for today's view
                            const reserved = bookings.find(b => b.roomNo === room.no && b.checkIn > today); 
                            if (reserved) status = 'reserved'; 
                        }
                    }

                    // Map colors and text colors
                    // status options: available, occupied, reserved, self-use
                    const colors = { 
                        'available': { bg: 'bg-emerald-50 border-emerald-200', text: 'text-emerald-700' }, 
                        'occupied': { bg: 'bg-red-50 border-red-200', text: 'text-red-700' }, 
                        'reserved': { bg: 'bg-amber-50 border-amber-200', text: 'text-amber-700' }, 
                        'self-use': { bg: 'bg-blue-50 border-blue-200', text: 'text-blue-700' } 
                    }; 
                    
                    // Default to available if invalid status
                    const style = colors[status] || colors['available'];

                    const card = document.createElement('div'); 
                    // Apply background and border classes
                    card.className = `room-card card p-1 border ${style.bg} flex flex-col items-center justify-center text-center relative ${isManual ? 'manual' : ''}`; 
                    card.style.minHeight = '50px'; 
                    
                    // Apply TEXT color to the room number
                    card.innerHTML = `
                        <span class="font-bold text-xs ${style.text}">${room.no}</span>
                        <span class="text-[8px] ${style.text} opacity-80">${status.toUpperCase()}</span>
                    `; 
                    
                    card.onclick = () => openRoomStatusModal(room.no);
                    grid.appendChild(card); 
                }); 
            }); 
        }

        function filterRooms(filter) { document.querySelectorAll('.room-filter-btn').forEach(btn => { btn.classList.remove('btn-primary'); btn.classList.add('btn-secondary'); if (btn.dataset.filter === filter) { btn.classList.remove('btn-secondary'); btn.classList.add('btn-primary'); } }); renderRooms(filter); }

        function handleFormSubmit(e) {
            if (e) e.preventDefault();
            const guestName = document.getElementById('guestName').value.trim(); const roomNo = document.getElementById('roomNo').value; const checkIn = document.getElementById('checkIn').value; const checkOut = document.getElementById('checkOut').value; const price = document.getElementById('price').value; const paymentStatus = document.getElementById('paymentStatus').value; const paymentMethod = document.getElementById('paymentMethod').value; const manualNoBooking = document.getElementById('noBooking').value.trim();
            if (!guestName || !roomNo || !checkIn || !checkOut || !price || !paymentStatus || !paymentMethod) { showToast('error', 'Validasi Gagal', 'Harap isi semua field wajib (*)'); return false; }
            const editIdx = parseInt(document.getElementById('editIndex').value); const nights = parseInt(document.getElementById('night').value) || 0; const priceVal = parseFloat(price) || 0;
            let total = priceVal * nights; if (paymentStatus === 'Compliment') total = 0;
            const finalNoBooking = manualNoBooking || generateBookingNo();
            const booking = { noBooking: finalNoBooking, bookingDate: document.getElementById('bookingDate').value, guestName: guestName, roomNo: roomNo, roomType: document.getElementById('roomType').value, checkIn: checkIn, checkOut: checkOut, night: nights, phone: document.getElementById('phone').value, price: priceVal, mealPlan: document.getElementById('mealPlan').value, qty: parseInt(document.getElementById('qty').value) || 1, paymentStatus: paymentStatus, paymentMethod: paymentMethod, reservationBy: document.getElementById('reservationBy').value, notes: '', paymentNotes: '', total: total };
            if (editIdx >= 0) { bookings[editIdx] = booking; showToast('success', 'Sukses', 'Booking berhasil diperbarui'); } else { bookings.push(booking); showToast('success', 'Sukses', 'Booking baru berhasil disimpan'); }
            saveBookings(); renderAll(); closeBookingModal(); return false;
        }
